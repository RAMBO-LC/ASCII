Database Schema (Supabase)
SQL

-- Users table (managed by Clerk, but we store extra data)
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  clerk_id TEXT UNIQUE NOT NULL,
  email TEXT,
  current_goal TEXT,
  streak_count INT DEFAULT 0,
  last_active DATE DEFAULT CURRENT_DATE,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Chat topics (separate conversations)
CREATE TABLE chat_topics (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES users(id),
  title TEXT NOT NULL,              -- e.g., "React Learning"
  tech_stack TEXT[],                 -- ['react', 'javascript']
  roadmap_ref TEXT,                  -- 'frontend' (from roadmap.sh)
  created_at TIMESTAMP DEFAULT NOW()
);

-- Messages
CREATE TABLE messages (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  topic_id UUID REFERENCES chat_topics(id) ON DELETE CASCADE,
  role TEXT CHECK (role IN ('user', 'assistant')),
  content TEXT NOT NULL,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Resources saved
CREATE TABLE saved_resources (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  topic_id UUID REFERENCES chat_topics(id),
  title TEXT,
  url TEXT,
  type TEXT,  -- 'video', 'article', 'course'
  created_at TIMESTAMP DEFAULT NOW()
);

'use client';

import { useState, useEffect } from 'react';
import { supabase } from '@/lib/supabase';
import TopicSidebar from './components/TopicSidebar';
import ChatWindow from './components/ChatWindow';

export default function ChatPage() {
  const [topics, setTopics] = useState([]);
  const [activeTopicId, setActiveTopicId] = useState(null);

  // Fetch user's chat topics
  useEffect(() => {
    fetchTopics();
  }, []);

  const fetchTopics = async () => {
    const { data } = await supabase
      .from('chat_topics')
      .select('*')
      .order('created_at', { ascending: false });
    setTopics(data || []);
  };

  const createNewTopic = async (title: string, techStack: string[]) => {
    const { data } = await supabase
      .from('chat_topics')
      .insert({ title, tech_stack: techStack })
      .select()
      .single();
    
    setTopics([data, ...topics]);
    setActiveTopicId(data.id);
  };

  return (
    <div className="flex h-screen bg-slate-900">
      {/* LEFT: Topic Sidebar */}
      <TopicSidebar
        topics={topics}
        activeId={activeTopicId}
        onSelect={setActiveTopicId}
        onNew={createNewTopic}
      />

      {/* RIGHT: Chat Window */}
      <ChatWindow topicId={activeTopicId} />
    </div>
  );
}

export default function TopicSidebar({ topics, activeId, onSelect, onNew }) {
  const [isCreating, setIsCreating] = useState(false);
  const [newTitle, setNewTitle] = useState('');

  const handleCreate = () => {
    if (!newTitle.trim()) return;
    onNew(newTitle, []); // Will detect tech stack from first message

    setIsCreating(false);
  };

  return (
    <div className="w-64 bg-slate-800 border-r border-slate-700 p-4">
      {/* New Topic Button */}
      <button
        onClick={() => setIsCreating(true)}
        className="w-full py-2 px-4 bg-blue-600 hover:bg-blue-700 rounded-lg mb-4"
      >
        + New Conversation
      </button>

      {/* Topic Creation Modal */}
      {isCreating && (
        <div className="mb-4">
          <input
            type="text"
            placeholder="Topic name (e.g., React Basics)"
            value={newTitle}
            onChange={(e) => setNewTitle(e.target.value)}
            className="w-full p-2 bg-slate-700 rounded mb-2"
          />
          <button onClick={handleCreate} className="w-full py-1 bg-green-600 rounded">
            Create
          </button>
        </div>
      )}

      {/* Topics List */}
      <div className="space-y-2">
        {topics.map((topic) => (

            key={topic.id}

'use client';

import { useState, useEffect, useRef } from 'react';
import { supabase } from '@/lib/supabase';

export default function ChatWindow({ topicId }) {
  const [messages, setMessages] = useState([]);
  const [input, setInput] = useState('');
  const [loading, setLoading] = useState(false);
  const messagesEndRef = useRef(null);

  // Fetch messages for active topic
  useEffect(() => {
    if (!topicId) return;
    fetchMessages();
  }, [topicId]);

  const fetchMessages = async () => {
    const { data } = await supabase
      .from('messages')
      .select('*')
      .eq('topic_id', topicId)
      .order('created_at', { ascending: true });
    setMessages(data || []);
  };

  const sendMessage = async () => {
    if (!input.trim() || !topicId) return;

    // Save user message
    const userMsg = { role: 'user', content: input, topic_id: topicId };
    await supabase.from('messages').insert(userMsg);
    setMessages([...messages, userMsg]);
    setInput('');
    setLoading(true);

    // Call AI
    const response = await fetch('/api/chat', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        topicId,
        message: input,
        history: messages, // Send context
      }),
    });

    const { reply } = await response.json();

    // Save AI response
    const aiMsg = { role: 'assistant', content: reply, topic_id: topicId };
    await supabase.from('messages').insert(aiMsg);
    setMessages((prev) => [...prev, aiMsg]);
    setLoading(false);
  };

  // Auto-scroll
  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  if (!topicId) {
    return (
      <div className="flex-1 flex items-center justify-center text-slate-500">
        Select a topic or create a new one
      </div>
    );
  }

  return (
    <div className="flex-1 flex flex-col">
      {/* Messages */}
      <div className="flex-1 overflow-y-auto p-6 space-y-4">
        {messages.map((msg, i) => (
          <div
            key={i}
            className={`flex ${msg.role === 'user' ? 'justify-end' : 'justify-start'}`}
          >
            <div
              className={`max-w-2xl p-4 rounded-lg ${
                msg.role === 'user'
                  ? 'bg-blue-600 text-white'
                  : 'bg-slate-700 text-slate-100'
              }`}
            >
              <ReactMarkdown>{msg.content}</ReactMarkdown>
            </div>
          </div>
        ))}
        {loading && <div className="text-slate-500">AI is typing...</div>}
        <div ref={messagesEndRef} />
      </div>

      {/* Input */}
      <div className="p-4 border-t border-slate-700">
        <div className="flex gap-2">
          <input
            type="text"
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyPress={(e) => e.key === 'Enter' && sendMessage()}
            placeholder="Ask anything..."
            className="flex-1 p-3 bg-slate-800 rounded-lg focus:ring-2 ring-blue-500"
          />
          <button
            onClick={sendMessage}
            disabled={loading}
            className="px-6 py-3 bg-blue-600 hover:bg-blue-700 rounded-lg disabled:opacity-50"
          >
            Send
          </button>
        </div>
      </div>
    </div>
  );
}
🔌 AI API Route - Memory + Context
app/api/chat/route.ts
TypeScript

import { NextRequest, NextResponse } from 'next/server';
import { model, SYSTEM_INSTRUCTION } from '@/lib/gemini';

export async function POST(req: NextRequest) {
  const { topicId, message, history } = await req.json();

  // Build context from history (last 10 messages)
  const context = history
    .slice(-10)
    .map((msg) => `${msg.role}: ${msg.content}`)
    .join('\n\n');

  const prompt = `
${SYSTEM_INSTRUCTION}

CONVERSATION HISTORY:
${context}

STUDENT'S NEW QUESTION:
${message}

YOUR RESPONSE (be specific, use examples):
`;

  try {
    const result = await model.generateContent(prompt);
    const reply = result.response.text();

    return NextResponse.json({ reply });
  } catch (error) {
    return NextResponse.json({ error: 'AI error' }, { status: 500 });
  }
}

🔥 Streak System
app/api/streak/route.ts
TypeScript

import { supabase } from '@/lib/supabase';
import { NextRequest, NextResponse } from 'next/server';

export async function POST(req: NextRequest) {
  const { userId } = await req.json();

  const { data: user } = await supabase
    .from('users')
    .select('*')
    .eq('clerk_id', userId)
    .single();

  const today = new Date().toISOString().split('T')[0];
  const lastActive = new Date(user.last_active).toISOString().split('T')[0];

  let newStreak = user.streak_count;

  // Check if yesterday
  const yesterday = new Date();
  yesterday.setDate(yesterday.getDate() - 1);
  const yesterdayStr = yesterday.toISOString().split('T')[0];

  if (lastActive === yesterdayStr) {
    newStreak += 1; // Continue streak
  } else if (lastActive !== today) {
    newStreak = 1; // Reset streak
  }

  // Update
  await supabase
    .from('users')
    .update({ streak_count: newStreak, last_active: today })
    .eq('clerk_id', userId);

  return NextResponse.json({ streak: newStreak });
}

import { NextRequest, NextResponse } from 'next/server';

export async function POST(req: NextRequest) {
  const { topic } = await req.json();

  // Use YouTube Data API (free tier: 10k requests/day)
  const response = await fetch(
    `https://www.googleapis.com/youtube/v3/search?part=snippet&q=${topic}+tutorial&type=video&maxResults=5&key=${process.env.YOUTUBE_API_KEY}`
  );

  const data = await response.json();
  return NextResponse.json(data.items);
}
